# Function to call GPT-4 BYOD
# def call_gpt4_byod(prompt_text, search_index_name, search_endpoint, filter_exp):
def call_gpt4_byod(prompt_text, search_index_name, search_endpoint,filter_exp):
    completion = client.chat.completions.create(
        model=deployment_id,
        messages=[
            {
                "role": "user",
                "content": prompt_text,
            },
        ],
        extra_body={
            "data_sources": [
                {
                    "type": "azure_search",
                    "parameters": {
                        "endpoint": search_endpoint,
                        "index_name": search_index_name,  
                        "authentication": {
                            "type": "api_key",
                            "key": search_api_key,
                        }
                    }
                }
            ],
        },
        # enhancements=None,
        temperature=0.7,
        top_p=1,
        max_tokens=800,
        stop=None,
        stream=False
        # seed=42
    )
    return completion
 
# Function to extract references from the completion response
def get_references(completion_response):
    # contexts = json.loads(completion_response["choices"][0]["message"]["context"]["messages"][0]["content"])
    # contexts = json.loads(completion_response.choices[0].message.context)
    contexts = completion_response.choices[0].message.context
    reference_tables = set()
    for cit in contexts["citations"]:
        reference_tables.add(cit["filepath"])
    return list(reference_tables)
