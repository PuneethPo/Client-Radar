technology_assessment_df = technology_assessment_df.select("Account", "ID", "Partner", "Product Description") \
    .filter(F.col("Partner").isin(*partner_mapping.keys())) \
    .withColumn("Partner", F.expr(f"CASE Partner {' '.join([f'WHEN \'{k}\' THEN \'{v}\'' for k, v in partner_mapping.items()])} END"))

technology_assessment_df = technology_assessment_df.filter(F.col("Product Description").isNotNull() & ~F.col("Product Description").startswith("zInactive"))

def get_all_products_top_partners(client_name):
    # Filter DataFrame for the specified client and valid partners
    filtered_df = technology_assessment_df.filter(F.col("Account") == client_name)

    # Early exit if no matching records
    if filtered_df.rdd.isEmpty():
        return f"No partner data found for {client_name}."

    # Clean the Product Description (remove "PARTNER - " prefix)
    cleaned_df = filtered_df.withColumn(
        "Clean Product Description",
        F.when(
            F.col("Product Description").contains(" - "),
            F.expr("split(`Product Description`, ' - ')[size(split(`Product Description`, ' - ')) - 1]")
        ).otherwise(F.col("Product Description"))
    )

    # Count product frequency per partner
    product_freq_df = cleaned_df.groupBy("Partner", "Clean Product Description").agg(F.count("*").alias("product_count"))

    # Define a window partitioned by Partner name, ordered by product frequency
    window_spec = Window.partitionBy("Partner").orderBy(F.desc("product_count"), F.asc("Clean Product Description"))

    # Rank products within each Partner and select top 12
    ranked_products_df = (
        product_freq_df.withColumn("rank", F.row_number().over(window_spec))
        .filter(F.col("rank") <= 12)
    )

    # Group by partner and collect list of cleaned product descriptions
    partner_products = (
        ranked_products_df.groupBy("Partner")
        .agg(F.collect_list("Clean Product Description").alias("Products"))
    )

    # Collect results
    partners_with_products_list = partner_products.collect()

    # Format result dictionary
    temp_dict = {}
    for row in partners_with_products_list:
        partner = row['Partner']
        products = [p for p in row['Products'] if p]  # Filter out nulls
        if products:
            temp_dict[partner] = products 

    # Final results
    result={}
    for partner in partners_list:
        if partner in temp_dict:
            result[partner] = temp_dict[partner]
    
    for partner in partners_list:
        if partner not in result:
            result[partner] = "Note: No data currently available"

    return result
