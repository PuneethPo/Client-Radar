# Create Empty Search Index Function

from azure.core.credentials import AzureKeyCredential

from azure.search.documents.indexes import SearchIndexClient  
from typing import List  
    
def create_index(name: str):  
    # Initialize the SearchIndexClient with endpoint and key  
    client = SearchIndexClient(  
        endpoint=search_service_endpoint,  
        credential=DefaultAzureCredential()  
    )  
  
    # Define the index fields  
    fields = [  
        SimpleField(name="id", type=SearchFieldDataType.String, key=True, retrievable=True, stored=True),  
        SimpleField(name="reference", type=SearchFieldDataType.String, filterable=True, retrievable=True, stored=True),  
        SearchableField(name="content", type=SearchFieldDataType.String, retrievable=True, stored=True),  
        SearchableField(name="client_account", type=SearchFieldDataType.String, filterable=True, retrievable=True, stored=True),  
        SearchableField(name="client_industry", type=SearchFieldDataType.String, filterable=True, retrievable=True, stored=True),  
        SearchableField(name="subsidiary_of", type=SearchFieldDataType.String, filterable=True, retrievable=True, stored=True),  
    ]  

    semantic_config = SemanticConfiguration(
    name="index-semantic-configuration",.
    prioritized_fields=SemanticPrioritizedFields(
        title_field=SemanticField(field_name="client_account"),
        keywords_fields=[
            SemanticField(field_name="client_account"),
            SemanticField(field_name="client_industry"),
            SemanticField(field_name="subsidiary_of")
        ],
        content_fields=[SemanticField(field_name="content")]
    )
    )

    semantic_search = SemanticSearch(configurations=[semantic_config])
  
    # Define CORS options  
    cors_options = CorsOptions(allowed_origins=["*"], max_age_in_seconds=60)  
  
    # Define the Search Index  
    index = SearchIndex(  
        name=name,  
        fields=fields,  
        semantic_search=semantic_search,
        cors_options=cors_options  
    )  
  
    # Create the index using the client  
    try:  
        result = client.create_index(index)  
        print(f"Index '{name}' created successfully.")  
    except Exception as e:  
        print(f"Error creating index '{name}': {e}")

# Create Indexer Function

def create_indexer(name: str, data_source_name: str, index_name: str):  
    client = SearchIndexerClient(  
        endpoint=search_service_endpoint,  
        credential=SearchIndexClient(  
        endpoint=search_service_endpoint,  
        credential=DefaultAzureCredential()  
    )  
    )  
  
    indexing_parameters = IndexingParameters(  
        configuration={  
            "parsingMode": "json",  
            "indexedDocumentFilenameExtensions": ".json"  
        }  
    )  
  
    indexer = SearchIndexer(  
        name=name,  
        data_source_name=data_source_name,  
        target_index_name=index_name,  
        parameters=indexing_parameters  
    )  
  
    try:  
        client.create_indexer(indexer)  
        print(f"Indexer '{name}' created successfully.")  
    except Exception as e:  
        print(f"Error creating indexer '{name}': {e}")

# Obtaining the Folder Path from Blob Storage

# Initialize the BlobServiceClient  
blob_service_client = BlobServiceClient.from_connection_string(conn_str)    
# Define the container name  
container_name = 'indexdatasource'  
# Get the container client  
container_client = blob_service_client.get_container_client(container_name)  
# List top-level folders in the container  
folder_names = set()  # Use a set to avoid duplicates  
# Getting the Folder Names from Blob
blob_list = container_client.walk_blobs(delimiter='/')  
for blob_prefix in blob_list:  
    # Each blob_prefix corresponds to a "directory"  
    folder_name = blob_prefix.name.rstrip('/')  
    folder_names.add(folder_name)  
# Save the unique top-level folder names to a variable  
unique_folder_names = sorted(folder_names)  
# Define a regex pattern to match the "YYYYMMDD" format  
date_pattern = re.compile(r'^\d{8}$')  
# Filter out folder names that match the "YYYYMMDD" pattern  
date_folders = [name for name in unique_folder_names if date_pattern.match(name)]  
# Convert these names to datetime objects  
date_objects = [datetime.strptime(date_str, "%Y%m%d") for date_str in date_folders]  
# Find the most recent date  
if date_objects:  
    latest_date = max(date_objects)  
    folder_path = latest_date.strftime("%Y%m%d")  
    print(folder_path)  
else:  
    print("No date folders found.")  
	
	

# Creating the Data Sources

def create_datasources_from_index_map(index_map, folder_path):  
    ds_client = SearchIndexerClient(  
        endpoint=search_service_endpoint,  
        credential=DefaultAzureCredential()  
    )  
      
    unique_values = set()  
    for value in index_map.values():  
        unique_values.update(value)  
      
    # Create a copy of the unique values set before normalization  
    unique_values_copy = unique_values.copy()  
  
    # Normalize the unique values by replacing underscores with dashes  
    unique_values_list = []  
    for value in unique_values:  
        normalized_value = re.sub(r'[^a-z0-9-]', '-', value.lower())  
        normalized_value = re.sub(r'-+', '-', normalized_value).strip('-')  
        normalized_value = normalized_value[:128]  
        unique_values_list.append(normalized_value)  
      
    # Iterate using the original and normalized values  
    for original_value, normalized_value in zip(unique_values_copy, unique_values_list):  
        datasource_path = f"{folder_path}/draup.{original_value}"  
        container = SearchIndexerDataContainer(name="indexes", query=datasource_path)  
        data_source_connection = SearchIndexerDataSourceConnection(  
            name=normalized_value,  
            type="azureblob",  
            connection_string=conn_str,  
            container=container  
        )  
        data_source = ds_client.create_data_source_connection(data_source_connection)  
        print(f"Data source '{normalized_value}' created successfully.")
